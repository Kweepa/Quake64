REM ** 6502 FAST BRESENHAM LINE ROUTINE
REM
REM    creation date: 2003-03-01  Michael Kircher
REM      last change: 2026-08-22  Michael Kircher
REM    - commentary amended from side documentation
REM    - introduced explicit labels for self-mod instructions
REM
REM    Licence: CC-BY-SA 4.0
REM
REM    This implementation of Bresenham on 6502 is supposed to plot
REM    into a 'font-like' bitmap, with the characters arranged in
REM    column-major order. All bytes within a 8-pixel bitmap column
REM    are accessed with the (),Y address mode of the 6502.
REM    
REM    The following draw actions apply:
REM
REM    zp_opcode <- &09, set pixels
REM    zp_opcode <- &29, clear pixels
REM    zp_opcode <- &49, invert pixels
:
zp_opcode=&B5
zp_error=&B6
zp_dx=&F7
zp_dy=&F8
zp_scrn=&F9
zp_x1=&FB
zp_y1=&FC
zp_x2=&FD
zp_y2=&FE
:
FOR pass=4 TO 7 STEP 3
P%=&3C00-2:O%=code
[OPT pass
 EQUW P% + 2

.Line
]

REM ** Sort endpoints (force x1 <= x2 for left-to-right drawing) and determine |dx|

[OPT pass
 SEC
 LDA zp_x2
 SBC zp_x1            ; calculate A = x2-x1
 BCS Line_00          ; if x2 >= x1, co-ordinates are already in order

 LDX zp_x1            ; swap x1/x2 using X and Y registers
 LDY zp_x2
 STY zp_x1
 STX zp_x2

 LDX zp_y1            ; swap y1/y2 along
 LDY zp_y2
 STY zp_y1
 STX zp_y2

 EOR #&FF             ; form 2's-complement of 'negative' difference in A
 ADC #&01             ; to obtain absolute value
.Line_00
 STA zp_dx            ; store |dx|
]

REM ** Determine vertical drawing direction and |dy|
 
[OPT pass 
 LDY #&C8             ; pre-load Y register with INY opcode (default down)
 SEC
 LDA zp_y2
 SBC zp_y1            ; calculate A = y2-y1
 BCS Line_01          ; if y2 >= y1, skip negation and keep INY

 LDY #&88             ; overwrite Y register with DEY opcode (up)
 EOR #&FF             ; form 2's-complement of 'negative' difference in A
 ADC #&01             ; to obtain absolute value
.Line_01
 STA zp_dy            ; store |dy|
]

REM ** Determine slope ('flat' vs. 'steep' Bresenham steps)

[OPT pass
 CMP zp_dx
 BCC Line_02          ; |dy|<|dx| => flat slope
 JMP Line_44          ; otherwise, steep slope (or 'diagonal')
]
